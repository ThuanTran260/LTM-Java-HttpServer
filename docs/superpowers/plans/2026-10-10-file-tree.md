    /**
     * Cay file web cua site, dang JSON. Chi lay file ma site phuc vu duoc (StaticFileHandler.contentTypeFor);
     * thu muc truoc, file sau, sap xep theo ten; thu muc khong co file web nao thi bo qua. Chi doc, khong sua file.
     */
    public synchronized String filesJson(String name) throws SiteException {
        Site s = require(name);
        try {
            return "[" + String.join(",", listWebFiles(folderOf(s), "")) + "]";
        } catch (IOException e) {
            throw new SiteException(500, "Khong doc duoc thu muc cua site: " + e.getMessage());
        }
    }

    /** Cac node JSON cua mot thu muc; prefix la duong dan tuong doi cua thu muc do (dung '/'). */
    private static List<String> listWebFiles(Path dir, String prefix) throws IOException {
        List<String> nodes = new ArrayList<>();
        if (!Files.isDirectory(dir)) {
            return nodes;
        }
        List<Path> entries;
        try (Stream<Path> stream = Files.list(dir)) {
            entries = stream.sorted(Comparator
                    .comparing((Path p) -> !Files.isDirectory(p, LinkOption.NOFOLLOW_LINKS))
                    .thenComparing(p -> p.getFileName().toString(), String.CASE_INSENSITIVE_ORDER))
                .toList();
        }
        for (Path p : entries) {
            String fileName = p.getFileName().toString();
            String rel = prefix.isEmpty() ? fileName : prefix + "/" + fileName;
            if (Files.isDirectory(p, LinkOption.NOFOLLOW_LINKS)) {
                List<String> children = listWebFiles(p, rel);
                if (!children.isEmpty()) {
                    nodes.add("{\"name\":\"" + jsonEscape(fileName) + "\",\"path\":\"" + jsonEscape(rel)
                        + "\",\"dir\":true,\"children\":[" + String.join(",", children) + "]}");
                }
            } else if (StaticFileHandler.contentTypeFor(fileName) != null) {
                nodes.add("{\"name\":\"" + jsonEscape(fileName) + "\",\"path\":\"" + jsonEscape(rel) + "\",\"dir\":false}");
            }
        }
        return nodes;
    }

    /** Escape chuoi cho JSON: dau nhay kep, dau gach cheo nguoc va ky tu dieu khien. */
    private static String jsonEscape(String s) {
        StringBuilder sb = new StringBuilder();
        for (char c : s.toCharArray()) {
            if (c == '"' || c == '\\') {
                sb.append('\\').append(c);
            } else if (c < 0x20) {
                sb.append(String.format("\\u%04x", (int) c));
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }

```js
async function toggleFiles(site, tr) {
  const open = tr.nextElementSibling;
  if (open && open.dataset.filesFor === site.name) {
    open.remove();
    return;
  }
  const row = document.createElement('tr');
  row.dataset.filesFor = site.name;
  const td = document.createElement('td');
  td.colSpan = 5;
  td.textContent = 'Đang tải danh sách file...';
  row.appendChild(td);
  tr.after(row);
  try {
    const nodes = await api('GET', '/api/sites/' + encodeURIComponent(site.name) + '/files');
    td.replaceChildren();
    if (!site.running) {
      const note = document.createElement('p');
      note.className = 'files-note';
      note.textContent = 'Web đang tắt: bấm Bật để mở các file bằng link bên dưới.';
      td.appendChild(note);
    }
    if (nodes.length === 0) {
      const empty = document.createElement('p');
      empty.textContent = 'Chưa có file web nào trong thư mục này.';
      td.appendChild(empty);
    } else {
      td.appendChild(buildTree(nodes, site.port));
    }
  } catch (e) {
    td.textContent = 'Không tải được danh sách file: ' + e.message;
  }
}

function buildTree(nodes, port) {
  const ul = document.createElement('ul');
  ul.className = 'tree';
  for (const n of nodes) {
    const li = document.createElement('li');
    if (n.dir) {
      const details = document.createElement('details');
      details.open = true;
      const summary = document.createElement('summary');
      summary.textContent = n.name + '/';
      details.appendChild(summary);
      details.appendChild(buildTree(n.children, port));
      li.appendChild(details);
    } else {
      const a = document.createElement('a');
      a.href = 'http://localhost:' + port + '/' + n.path.split('/').map(encodeURIComponent).join('/');
      a.target = '_blank';
      a.rel = 'noopener';
      a.textContent = n.name;
      li.appendChild(a);
    }
    ul.appendChild(li);
  }
  return ul;
}

```
